  // 行程时间节点定义 (带 GMT+0800 时区)

  const scheduleEvents = [

    { title: "川航 3U3911 起飞 (成都天府)", time: "2026-09-12T20:05:00+08:00" },

    { title: "包车前往怡保", time: "2026-09-16T09:30:00+08:00" },

    { title: "包车前往吉隆坡", time: "2026-09-17T14:00:00+08:00" },

    { title: "亚航 AK6218 起飞去登嘉楼", time: "2026-09-18T16:10:00+08:00" },

    { title: "登船前往热浪岛彩虹度假村", time: "2026-09-19T08:30:00+08:00" },

    { title: "亚航 AK6219 返回吉隆坡", time: "2026-09-21T17:45:00+08:00" },

    { title: "川航 3U3994 返程成都", time: "2026-09-24T00:20:00+08:00" }

  ];


  function updateCountdown() {

    const now = new Date().getTime();

    let nextEvent = null;


    for (let ev of scheduleEvents) {

      const evTime = new Date(ev.time).getTime();

      if (evTime > now) {

        nextEvent = { title: ev.title, target: evTime };

        break;

      }

    }


    if (!nextEvent) {

      document.getElementById("nextEventTitle").innerText = "行程进行中 / 已结束";

      document.getElementById("d").innerText = "00";

      document.getElementById("h").innerText = "00";

      document.getElementById("m").innerText = "00";

      document.getElementById("s").innerText = "00";

      return;

    }


    document.getElementById("nextEventTitle").innerText = nextEvent.title;

    const diff = nextEvent.target - now;


    const days = Math.floor(diff / (1000 * 60 * 60 * 24));

    const hours = Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));

    const minutes = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60));

    const seconds = Math.floor((diff % (1000 * 60)) / 1000);


    document.getElementById("d").innerText = String(days).padStart(2, '0');

    document.getElementById("h").innerText = String(hours).padStart(2, '0');

    document.getElementById("m").innerText = String(minutes).padStart(2, '0');

    document.getElementById("s").innerText = String(seconds).padStart(2, '0');

  }


  setInterval(updateCountdown, 1000);

  updateCountdown();

</script>
